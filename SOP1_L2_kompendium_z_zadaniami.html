ROZWIĄZANIE · C
Kompletny przykład
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <sys/stat.h>
#include <stdio.h>

int main(void) {
    int fd = open("map.txt", O_RDWR);
    if (fd == -1) { perror("open"); return 1; }
    struct stat st;
    if (fstat(fd, &st) == -1) { perror("fstat"); close(fd); return 1; }
    if (st.st_size == 0) {
        fprintf(stderr, "Pusty plik: nic do mapowania\n");
        close(fd); return 1;
    }
    void *mem = mmap(NULL, (size_t)st.st_size,
                     PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (mem == MAP_FAILED) { perror("mmap"); close(fd); return 1; }
    ((char *)mem)[0] = 'X';
    int error = 0;
    if (msync(mem, (size_t)st.st_size, MS_SYNC) == -1) {
        perror("msync"); error = 1;
    }
    if (munmap(mem, (size_t)st.st_size) == -1) {
        perror("munmap"); error = 1;
    }
    if (close(fd) == -1) { perror("close"); error = 1; }
    return error;
}

#include <poll.h>
#include <unistd.h>
#include <errno.h>
#include <stdio.h>

int main(void) {
    struct pollfd pfd = {.fd = STDIN_FILENO, .events = POLLIN, .revents = 0};

    if (pfd.revents & (POLLIN | POLLHUP)) {

        if (n == -1) { perror("read"); return 1; }
        if (n == 0) { puts("EOF"); return 0; }
        if (write(STDOUT_FILENO, buf, (size_t)n) != n) {
            perror("write"); return 1;
        }
    }
    return 0;
}

Co naprawdę jest w L2?

↑

Oparte na materiałach 2026 w repozytorium SOP-MINI, nie na starszej wersji witryny z inną numeracją.

A · obowiązkowa baza

open, close, read, write, lseek, flagi, uprawnienia, deskryptor.

B · sedno laboratorium

poll, fcntl, O_NONBLOCK, FIFO, EAGAIN, readv, writev, mmap.

    while (len) {
        ssize_t n = write(fd, p, len);

int main(int argc, char **argv) {
    if (argc != 3) {
        fprintf(stderr, "Użycie: %s src dst\n", argv[0]); return 1;
    }
    int src = open(argv[1], O_RDONLY);

    if (stat(argv[2], &b) == 0) {
        if (a.st_dev == b.st_dev && a.st_ino == b.st_ino) {
            fprintf(stderr, "Źródło i cel to ten sam plik\n");
            close(src); return 1;
